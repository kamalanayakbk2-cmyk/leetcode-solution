class Solution:
    def reverse(self, x: int) -> int:
        # Define 32-bit signed integer boundaries
        INT_MIN, INT_MAX = -2**31, 2**31 - 1
        
        # Determine the sign and work with the absolute value
        sign = -1 if x < 0 else 1
        x = abs(x)
        
        # Reverse the digits using string slicing
        reversed_x = sign * int(str(x)[::-1])
        
        # Return 0 if the reversed value falls outside the 32-bit range
        if reversed_x < INT_MIN or reversed_x > INT_MAX:
            return 0
            
        return reversed_x
